package BinaryTree;

import java.util.*;

public class PreorderTree {
    static class Node {
        int data;
        Node left;
        Node right;

        Node(int data) {
            this.data = data;
            this.left = null;
            this.right = null;
        }
    }

    static class BinaryTree {
        static int idx = -1;

        // { 1, 2, 4, -1, -1, 5, -1, -1, 3, -1, 6, -1, -1 };
        public static Node BinaryTreePreorder(int nodes[]) {
            idx++;
            if (nodes[idx] == -1) {
                return null;
            }
            Node newnode = new Node(nodes[idx]);
            newnode.left = BinaryTreePreorder(nodes);
            newnode.right = BinaryTreePreorder(nodes);

            return newnode;
        }

        public static void preorder(Node root) {
            if (root == null) {
                System.out.print("-1" + " ");
                return;
            }
            System.out.print(root.data + " ");
            preorder(root.left);
            preorder(root.right);
        }

        public static void inorder(Node root) {
            if (root == null) {
                System.out.print("-1" + " ");
                return;
            }
            inorder(root.left);
            System.out.print(root.data + " ");
            inorder(root.right);

        }

        public static void postorder(Node root) {
            if (root == null) {
                System.out.print("-1" + " ");
                return;
            }
            postorder(root.left);
            postorder(root.right);
            System.out.print(root.data + " ");

        }

        public static int heigth(Node root) {
            if (root == null) {
                return 0;
            }
            int treeHeight;
            int lh = heigth(root.left);
            int rh = heigth(root.right);
            treeHeight = Math.max(lh, rh) + 1;

            return treeHeight;
        }

        public static int daimeter2(Node root) {
            if (root == null) {
                return 0;
            }
            int maxDiameter;
        
            int ld = daimeter2(root.left);
            int lh = heigth(root.left);
            int rd = daimeter2(root.right);
            int rh = heigth(root.right);

            int seldaimeter = lh + rh + 1;
            maxDiameter = Math.max(Math.max(ld, rd) ,seldaimeter);

            return maxDiameter;
        }


        public static int countNodes(Node root) {
            if (root == null) {
                return 0;
            }
            int lcount = countNodes(root.left);
            int rcount = countNodes(root.right);
            int totalCount = 0;
            totalCount = lcount + rcount + 1;

            return totalCount;
        }

        public static int sumOfNodes(Node root) {
            if (root == null) {
                return 0;
            }
            int lsum = sumOfNodes(root.left);
            int rsum = sumOfNodes(root.right);
            int totalsum = 0;
            totalsum = lsum + rsum + root.data;

            return totalsum;
        }



         static class info{
            int daim;
            int hei;

            info(int d,int h){
                this.daim=d;
                this.hei=h;
            }
     }

        public static int daimeter(Node root) {
            if (root == null) {
                return 0;
            }
            int maxDiameter;
        
            info leftinfo = daimeter(root.left);
            info rightinfo = daimeter(root.right);

            int daimeter= Math.max(Math.max(leftinfo.daim, rightinfo.daim),leftinfo.hei+ rightinfo.hei+1);
            int  height=Math.max(leftinfo.hei,rightinfo.hei)+1;


            return new info(daimeter, height);
        
        }

        public static void levelOrder(Node root) {
            if (root == null) {
                return;
            }
            Queue<Node> q = new LinkedList<>();
            q.add(root);
            q.add(null); // to print new line

            while (!q.isEmpty()) {
                Node curr = q.remove();
                if (curr == null) {
                    System.out.println();
                    if (q.isEmpty()) {
                        break;
                    } else {
                        q.add(null);
                    }
                } else {
                    System.out.print(curr.data + " ");
                    if (curr.left != null) {
                        q.add(curr.left);
                    }
                    if (curr.right != null) {
                        q.add(curr.right);
                    }
                }
            }
        }
    }
